cmake_minimum_required(VERSION 3.20)
project(tradeapi_demo LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build for Windows x86; for Visual Studio use -A Win32")
endif()
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
add_executable(tradeapi_demo main.cpp)
target_compile_features(tradeapi_demo PRIVATE cxx_std_17)
target_link_libraries(tradeapi_demo PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(tradeapi_demo PRIVATE /W4 /utf-8)
endif()
